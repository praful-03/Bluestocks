-- 1. BASE OPERATIONAL MEASURES
Total Sales = SUM(fact_sales[Sales])
Total Profit = SUM(fact_sales[Profit])
Profit Margin % = DIVIDE([Total Profit], [Total Sales], 0)
Total Orders = DISTINCTCOUNT(fact_sales[Order_ID])
Total Units Sold = SUM(fact_sales[Quantity])
Average Order Value = DIVIDE([Total Sales], [Total Orders], 0)

-- 2. TIME INTELLIGENCE & YOY COMPARISONS
Sales YTD = TOTALYTD([Total Sales], dim_date[Date])
Sales Prior Year = CALCULATE([Total Sales], SAMEPERIODLASTYEAR(dim_date[Date]))
YoY Sales Growth % = 
VAR PrevYear = [Sales Prior Year]
VAR CurrYear = [Total Sales]
RETURN
    IF(ISBLANK(PrevYear), BLANK(), DIVIDE(CurrYear - PrevYear, PrevYear, 0))

-- 3. MOVING AVERAGE
Sales 3M Moving Avg = 
AVERAGEX(
    DATESINPERIOD(dim_date[Date], LASTDATE(dim_date[Date]), -3, MONTH),
    [Total Sales]
)

-- 4. ADVANCED SEGMENTATION & TOP N
Top 5 Products Sales = 
CALCULATE(
    [Total Sales],
    KEEPFILTERS(TOPN(5, ALLSELECTED(dim_products[Product]), [Total Sales], DESC))
)

Margin Health Status = 
SWITCH(
    TRUE(),
    [Profit Margin %] >= 0.20, "High Margin (>20%)",
    [Profit Margin %] >= 0.10, "Target Margin (10-20%)",
    [Profit Margin %] >= 0.00, "Low Margin (0-10%)",
    "Negative Margin (Loss)"
)
